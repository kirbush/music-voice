cmake_minimum_required(VERSION 3.22)
project(musicvoice_futo LANGUAGES C CXX)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)

add_library(musicvoice_futo SHARED futo_jni.cpp)
target_compile_features(musicvoice_futo PRIVATE cxx_std_17)
target_compile_options(musicvoice_futo PRIVATE -fvisibility=hidden)
target_link_options(musicvoice_futo PRIVATE -static-libstdc++ -Wl,--exclude-libs,ALL -Wl,--no-undefined)
target_link_libraries(musicvoice_futo PRIVATE whisper)
